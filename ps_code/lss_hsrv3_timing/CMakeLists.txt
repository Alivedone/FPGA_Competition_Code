cmake_minimum_required(VERSION 3.16)
project(lss_hsrv3_ps_final LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LSS_WITH_ICRAFT "Build Icraft hardware backend" ON)
find_package(Threads REQUIRED)
add_library(ps_hotpath STATIC src/ps_hotpath.cpp)
set_property(TARGET ps_hotpath PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
if(MSVC)
  target_compile_options(ps_hotpath PRIVATE /O2 /fp:precise /GL-)
else()
  target_compile_options(ps_hotpath PRIVATE -O2 -fno-fast-math -fno-lto)
endif()
add_library(hmi_module src/hmi_history.cpp src/hmi_uart.cpp)
target_link_libraries(hmi_module PUBLIC Threads::Threads ps_hotpath)
add_executable(lss_v3_timing src/main.cpp)
target_link_libraries(lss_v3_timing PRIVATE hmi_module)
if(WIN32)
  target_link_libraries(lss_v3_timing PRIVATE ws2_32)
endif()
if(LSS_WITH_ICRAFT)
  find_package(Icraft-ZG330Backend REQUIRED)
  find_package(Icraft-HostBackend REQUIRED)
  target_compile_definitions(lss_v3_timing PRIVATE LSS_WITH_ICRAFT=1)
  target_link_libraries(lss_v3_timing PRIVATE Icraft::ZG330Backend Icraft::HostBackend)
endif()
foreach(target ps_hotpath hmi_module lss_v3_timing)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /Zc:__cplusplus)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
    if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
      target_link_libraries(${target} PRIVATE stdc++fs)
    endif()
  endif()
endforeach()
